Task 8: Customer Feedback Analysis
# Task 8
def count_vowels(text: str, include_y: bool = True) -> int:
    vowels = set("aeiouAEIOU" + ("yY" if include_y else ""))
    return sum(1 for ch in text if ch in vowels)

def reverse_text(text: str) -> str:
    return text[::-1]

feedback = "The service was excellent and very quick!"

print("Customer Feedback Analysis:")
print("Original Feedback:", feedback)
print("Number of Vowels:", count_vowels(feedback, include_y=True))  # -> 13
print("Reversed Feedback:", reverse_text(feedback))

Task 12: Discount Calculation
# Task 12
products = [
    {"name": "Product A", "original_price": 100, "discount": 10},
    {"name": "Product B", "original_price": 250, "discount": 20},
    {"name": "Product C", "original_price": 75, "discount": 15},
    {"name": "Product D", "original_price": 150, "discount": 5}
]

    final = p["original_price"] * (1 - p["discount"]/100)
    print(f"Product {i}: Original = ${p['original_price']}, "
          f"Discount = {p['discount']}%, Final = ${round(final,2)}")

Task 13: Customer Feedback Sentiment Analysis
# Task 13
feedback = "I am very happy with the service. It was a good experience!"
positives = ["good", "happy", "excellent", "great"]
negatives = ["bad", "disappointed", "poor", "terrible"]

text = feedback.lower()
sentiment = "Neutral"

for w in positives:
    if w in text:
        sentiment = "Positive"

for w in negatives:
    if w in text:
        sentiment = "Negative"

print("Sentiment:", sentiment)

Task 14: Employee Salary Increment Calculator
# Task 14
employees = {
    "Alice": {"salary": 50000, "rating": "Excellent"},
    "Bob": {"salary": 40000, "rating": "Good"},
    "Charlie": {"salary": 45000, "rating": "Average"},
    "David": {"salary": 35000, "rating": "Poor"}
}

increments = {"Excellent": 20, "Good": 15, "Average": 10, "Poor": 5}

for name, info in employees.items():
    pct = increments[info["rating"]]
    new_salary = info["salary"] * (1 + pct/100)
    print(f"{name}: Updated Salary = ${round(new_salary,2)}")
